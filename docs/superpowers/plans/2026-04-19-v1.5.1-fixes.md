```swift
import Foundation

public struct GuestMigrationResult: Sendable {
    public let workoutsMigrated: Int
    public let challengesMigrated: Int
    public let injuriesMigrated: Int
    public let oneRepMaxesMigrated: Int
    public let benchmarkResultsMigrated: Int
    public let programsMigrated: Int
    public let settingsMigrated: Int
}

public actor GuestDataMigrator {
    private let source: any DataClientProtocol
    private let destination: any DataClientProtocol

    public init(source: any DataClientProtocol, destination: any DataClientProtocol) {
        self.source = source
        self.destination = destination
    }

    public func migrate() async throws -> GuestMigrationResult {
        let workouts: [Workout] = try await source.fetchAll(recordType: "Workout")
        let challenges: [Challenge] = try await source.fetchAll(recordType: "Challenge")
        let injuries: [Injury] = try await source.fetchAll(recordType: "Injury")
        let ormRecords: [OneRepMaxRecord] = try await source.fetchAll(recordType: "OneRepMaxRecord")
        let benchmarkResults: [BenchmarkResult] = try await source.fetchAll(recordType: "BenchmarkResult")
        let programs: [EnrolledProgramRecord] = try await source.fetchAll(recordType: "EnrolledProgram")
        let settings: [UserSettingsRecord] = try await source.fetchAll(recordType: "UserSettings")

        // Save to destination. If any save throws, we abort without clearing source.
        if !workouts.isEmpty        { try await destination.save(workouts, recordType: "Workout") }
        if !challenges.isEmpty      { try await destination.save(challenges, recordType: "Challenge") }
        if !injuries.isEmpty        { try await destination.save(injuries, recordType: "Injury") }
        if !ormRecords.isEmpty      { try await destination.save(ormRecords, recordType: "OneRepMaxRecord") }
        if !benchmarkResults.isEmpty{ try await destination.save(benchmarkResults, recordType: "BenchmarkResult") }
        if !programs.isEmpty        { try await destination.save(programs, recordType: "EnrolledProgram") }
        if !settings.isEmpty        { try await destination.save(settings, recordType: "UserSettings") }

        // Only clear source if every save succeeded.
        try await source.deleteAll(recordType: "Workout")
        try await source.deleteAll(recordType: "Challenge")
        try await source.deleteAll(recordType: "Injury")
        try await source.deleteAll(recordType: "OneRepMaxRecord")
        try await source.deleteAll(recordType: "BenchmarkResult")
        try await source.deleteAll(recordType: "EnrolledProgram")
        try await source.deleteAll(recordType: "UserSettings")

        return GuestMigrationResult(
            workoutsMigrated: workouts.count,
            challengesMigrated: challenges.count,
            injuriesMigrated: injuries.count,
            oneRepMaxesMigrated: ormRecords.count,
            benchmarkResultsMigrated: benchmarkResults.count,
            programsMigrated: programs.count,
            settingsMigrated: settings.count
        )
    }
}
```
