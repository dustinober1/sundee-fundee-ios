/// Start a training cycle for the test user.
/// Must be called after completeOnboarding.
Future<void> startCycle(
  WidgetTester tester, {
  String programKey = 'program-back-squat-complete-cycle',
}) async {
  // Navigate to programs
  await tester.tap(find.byKey(const Key('nav-programs')));
  await tester.pumpAndSettle();
  expect(find.byKey(const Key('programs-screen')), findsOneWidget);

  // Tap the program to start a cycle
  await tester.tap(find.byKey(Key(programKey)));
  await tester.pumpAndSettle();

  // If start cycle dialog appears, confirm it
  final startButton = find.byKey(const Key('start-cycle-button'));
  if (startButton.evaluate().isNotEmpty) {
    await tester.tap(startButton);
    await tester.pumpAndSettle();
  }

  // Should now be on workout screen
  expect(find.byKey(const Key('workout-screen')), findsOneWidget);
}

/// Navigate to an active workout (assumes cycle already started).
Future<void> navigateToWorkout(
  WidgetTester tester, {
  String programKey = 'program-back-squat-complete-cycle',
}) async {
  // Navigate to programs
  await tester.tap(find.byKey(const Key('nav-programs')));
  await tester.pumpAndSettle();

  // Tap the program
  await tester.tap(find.byKey(Key(programKey)));
  await tester.pumpAndSettle();

  expect(find.byKey(const Key('workout-screen')), findsOneWidget);
}

/// Log a set with given weight and reps.
Future<void> logSet(
  WidgetTester tester, {
  required int setNumber,
  required String weight,
  required String reps,
}) async {
  // Enter weight
  await tester.enterText(
    find.byKey(Key('set-$setNumber-weight-input')),
    weight,
  );
  // Enter reps
  await tester.enterText(
    find.byKey(Key('set-$setNumber-reps-input')),
    reps,
  );
  // Tap log button
  await tester.tap(find.byKey(Key('set-$setNumber-log-button')));
  await tester.pumpAndSettle();
}

/// Dismiss rest timer sheet if visible.
Future<void> dismissRestTimer(WidgetTester tester) async {
  // Check if rest timer sheet is visible
  final skipButton = find.byKey(const Key('rest-timer-skip-button'));
  if (skipButton.evaluate().isNotEmpty) {
    await tester.tap(skipButton);
    await tester.pumpAndSettle();
    // Wait for sheet to dismiss
    await tester.pump(const Duration(milliseconds: 500));
  }
}

testWidgets('completes workout while offline and persists locally (WORK-03)',
    (tester) async {
  // Start offline
  fakeConnectivity.goOffline();
  await pumpApp(tester);
  await completeOnboarding(tester);

  // Verify offline banner
  await tester.pump(const Duration(milliseconds: 500));
  await tester.pumpAndSettle();
  expect(find.byKey(const Key('offline-banner')), findsOneWidget);

  // Navigate to workout
  await startCycle(tester);

  // Log set while offline
  await logSet(tester, setNumber: 1, weight: '135', reps: '5');
  await dismissRestTimer(tester);

  // Complete workout
  await tester.tap(find.byKey(const Key('complete-workout-button')));
  await tester.pumpAndSettle();

  // Should return to dashboard
  expect(find.byKey(const Key('dashboard-screen')), findsOneWidget);

  // Verify Drift persistence (offline, local storage)
  final container = ProviderScope.containerOf(
    tester.element(find.byKey(const Key('dashboard-screen'))),
  );
  final db = container.read(databaseProvider);

  final workouts = await db.select(db.completedWorkouts).get();
  expect(
    workouts,
    hasLength(1),
    reason: 'Workout should persist locally even when offline',
  );
  expect(workouts.first.programId, 'back-squat-complete-cycle');

  final sets = await (db.select(db.completedSets)
        ..where((t) => t.workoutId.equals(workouts.first.id)))
      .get();
  expect(sets, hasLength(1), reason: 'Set should persist locally');
  expect(sets.first.actualWeight, 135);

  // Reconnect and verify data still exists
  fakeConnectivity.goOnline();
  await tester.pump(const Duration(milliseconds: 500));
  await tester.pumpAndSettle();

  // Offline banner should be gone
  expect(find.byKey(const Key('offline-banner')), findsNothing);

  // Data should still exist after reconnect
  final workoutsAfterReconnect = await db.select(db.completedWorkouts).get();
  expect(
    workoutsAfterReconnect,
    hasLength(1),
    reason: 'Workout should persist through reconnect',
  );
});
