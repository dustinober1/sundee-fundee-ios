'use client';

import { useState } from 'react';
import { useUser } from '@/contexts/user-context';
import {
  Popover,
  PopoverContent,
  PopoverTrigger,
} from '@/components/ui/popover';
import { Button } from '@/components/ui/button';
import type { SyncStatus } from '@/types/user';

function formatTimeAgo(date: Date): string {
  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
  
  if (seconds < 60) return 'just now';
  if (seconds < 120) return '1 min ago';
  if (seconds < 3600) return `${Math.floor(seconds / 60)} min ago`;
  if (seconds < 7200) return '1 hour ago';
  if (seconds < 86400) return `${Math.floor(seconds / 3600)} hours ago`;
  return `${Math.floor(seconds / 86400)} days ago`;
}

function SyncStatusIcon({ status }: { status: SyncStatus }) {
  switch (status) {
    case 'synced':
      return (
        <svg className="h-4 w-4 text-green-500" fill="none" viewBox="0 0 24 24" stroke="currentColor">
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M5 13l4 4L19 7" />
        </svg>
      );
    case 'syncing':
      return (
        <svg className="h-4 w-4 text-blue-500 animate-spin" fill="none" viewBox="0 0 24 24" stroke="currentColor">
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M4 4v5h.582m15.356 2A8.001 8.001 0 004.582 9m0 0H9m11 11v-5h-.581m0 0a8.003 8.003 0 01-15.357-2m15.357 2H15" />
        </svg>
      );
    case 'offline':
      return (
        <svg className="h-4 w-4 text-gray-400" fill="none" viewBox="0 0 24 24" stroke="currentColor">
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M18.364 5.636a9 9 0 010 12.728m0 0l-2.829-2.829m2.829 2.829L21 21M15.536 8.464a5 5 0 010 7.072m0 0l-2.829-2.829m-4.243 2.829a4.978 4.978 0 01-1.414-2.83m-1.414 5.658a9 9 0 01-2.167-9.238m7.824 2.167a1 1 0 111.414 1.414m-1.414-1.414L3 3m8.293 8.293l1.414 1.414" />
        </svg>
      );
    case 'pending':
      return (
        <svg className="h-4 w-4 text-yellow-500" fill="none" viewBox="0 0 24 24" stroke="currentColor">
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 8v4l3 3m6-3a9 9 0 11-18 0 9 9 0 0118 0z" />
        </svg>
      );
    case 'error':
      return (
        <svg className="h-4 w-4 text-red-500" fill="none" viewBox="0 0 24 24" stroke="currentColor">
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 9v2m0 4h.01m-6.938 4h13.856c1.54 0 2.502-1.667 1.732-3L13.732 4c-.77-1.333-2.694-1.333-3.464 0L3.34 16c-.77 1.333.192 3 1.732 3z" />
        </svg>
      );
    case 'disabled':
      return (
        <svg className="h-4 w-4 text-gray-300" fill="none" viewBox="0 0 24 24" stroke="currentColor">
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M7 16a4 4 0 01-.88-7.903A5 5 0 1115.9 6L16 6a5 5 0 011 9.9M15 13l-3-3m0 0l-3 3m3-3v12" />
        </svg>
      );
    default:
      return null;
  }
}

function SyncStatusLabel({ status }: { status: SyncStatus }) {
  switch (status) {
    case 'synced':
      return <span className="text-green-600">Synced</span>;
    case 'syncing':
      return <span className="text-blue-600">Syncing...</span>;
    case 'offline':
      return <span className="text-gray-500">Offline</span>;
    case 'pending':
      return <span className="text-yellow-600">Pending</span>;
    case 'error':
      return <span className="text-red-600">Error</span>;
    case 'disabled':
      return <span className="text-gray-400">Sync off</span>;
    default:
      return null;
  }
}

export function SyncStatusPopover() {
  const { 
    syncStatus, 
    lastSyncedAt, 
    isAuthenticated, 
    isOnline,
    pullFromCloud 
  } = useUser();
  const [isSyncing, setIsSyncing] = useState(false);

  const handleSyncNow = async () => {
    if (!isOnline || !isAuthenticated) return;
    setIsSyncing(true);
    try {
      await pullFromCloud();
    } finally {
      setIsSyncing(false);
    }
  };

  const pendingCount = typeof window !== 'undefined' 
    ? JSON.parse(localStorage.getItem('sync_pending_workout_ids') ?? '[]').length 
    : 0;

  return (
    <Popover>
      <PopoverTrigger asChild>
        <button 
          className="flex items-center gap-1.5 rounded-md px-2 py-1 text-sm hover:bg-muted transition-colors"
          aria-label="Sync status"
        >
          <SyncStatusIcon status={syncStatus} />
          <SyncStatusLabel status={syncStatus} />
        </button>
      </PopoverTrigger>
      <PopoverContent className="w-64" align="end">
        <div className="space-y-3">
          <div className="flex items-center justify-between">
            <span className="font-medium">Sync Status</span>
            <div className="flex items-center gap-1">
              <SyncStatusIcon status={syncStatus} />
              <SyncStatusLabel status={syncStatus} />
            </div>
          </div>

          {lastSyncedAt && (
            <div className="text-sm text-muted-foreground">
              Last synced: {formatTimeAgo(lastSyncedAt)}
            </div>
          )}

          {pendingCount > 0 && (
            <div className="text-sm text-yellow-600">
              {pendingCount} workout{pendingCount > 1 ? 's' : ''} waiting to sync
            </div>
          )}

          {!isOnline && (
            <div className="text-sm text-muted-foreground">
              You&apos;re offline. Data will sync when you reconnect.
            </div>
          )}

          {isAuthenticated && isOnline && (
            <Button 
              variant="outline" 
              size="sm" 
              className="w-full"
              onClick={handleSyncNow}
              disabled={isSyncing || syncStatus === 'syncing'}
            >
              {isSyncing || syncStatus === 'syncing' ? 'Syncing...' : 'Sync now'}
            </Button>
          )}

          {!isAuthenticated && (
            <p className="text-xs text-muted-foreground">
              Sign in to enable cloud sync
            </p>
          )}
        </div>
      </PopoverContent>
    </Popover>
  );
}
```

This component:
- Shows sync status icon and label as trigger
- Opens popover with sync details
- Shows last synced timestamp in human-readable format
- Shows pending workout count if any
- Shows offline message when disconnected
- Provides "Sync now" button when authenticated and online

'use client';

import { useState } from 'react';
import { useUser } from '@/contexts/user-context';
import { SyncStatusPopover } from './sync-status-popover';
import { AuthDialog } from '@/components/auth/auth-dialog';
import { Button } from '@/components/ui/button';
import {
  Popover,
  PopoverContent,
  PopoverTrigger,
} from '@/components/ui/popover';

function UserAvatar({ name }: { name: string }) {
  const initials = name
    .split(' ')
    .map(n => n[0])
    .join('')
    .toUpperCase()
    .slice(0, 2);

  return (
    <div className="flex h-8 w-8 items-center justify-center rounded-full bg-primary text-primary-foreground text-sm font-medium">
      {initials}
    </div>
  );
}

export function DashboardHeader() {
  const { user, isAuthenticated, authUser, signOut } = useUser();
  const [showAuthDialog, setShowAuthDialog] = useState(false);

  const displayName = user?.name ?? 'User';
  const email = authUser?.email;

  return (
    <>
      <header className="flex items-center justify-between py-2">
        <h1 className="text-2xl font-bold">Dashboard</h1>
        
        <div className="flex items-center gap-3">
          <SyncStatusPopover />
          
          {isAuthenticated ? (
            <Popover>
              <PopoverTrigger asChild>
                <button className="flex items-center gap-2 rounded-md px-2 py-1 hover:bg-muted transition-colors">
                  <UserAvatar name={displayName} />
                  <span className="text-sm font-medium hidden sm:inline">{displayName}</span>
                </button>
              </PopoverTrigger>
              <PopoverContent className="w-56" align="end">
                <div className="space-y-3">
                  <div>
                    <p className="font-medium">{displayName}</p>
                    {email && (
                      <p className="text-sm text-muted-foreground truncate">{email}</p>
                    )}
                  </div>
                  <Button 
                    variant="outline" 
                    size="sm" 
                    className="w-full"
                    onClick={signOut}
                  >
                    Sign out
                  </Button>
                </div>
              </PopoverContent>
            </Popover>
          ) : (
            <Button 
              variant="outline" 
              size="sm"
              onClick={() => setShowAuthDialog(true)}
            >
              Sign in
            </Button>
          )}
        </div>
      </header>

      <AuthDialog 
        open={showAuthDialog} 
        onOpenChange={setShowAuthDialog} 
      />
    </>
  );
}
```
