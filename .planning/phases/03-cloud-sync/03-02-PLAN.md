'use client';

import { useEffect, useCallback, useState, useRef } from 'react';
import { createClient } from '@/lib/supabase/client';
import { 
  pushWorkout, 
  pullLatest, 
  uploadAllLocalData,
  withRetry,
  getQueue, 
  enqueueWorkout, 
  dequeueWorkout,
  clearQueue
} from '@/lib/sync';
import { useOnlineStatus } from './use-online-status';
import type { SyncStatus } from '@/types/user';
import type { User as SupabaseUser } from '@supabase/supabase-js';

interface UseSyncReturn {
  syncStatus: SyncStatus;
  lastSyncedAt: Date | null;
  authUser: SupabaseUser | null;
  isAuthenticated: boolean;
  syncAfterWorkout: (workoutId: string) => Promise<void>;
  pullFromCloud: () => Promise<void>;
  uploadExistingData: () => Promise<boolean>;
  signOut: () => Promise<void>;
}

export function useSync(): UseSyncReturn {
  const isOnline = useOnlineStatus();
  const [syncStatus, setSyncStatus] = useState<SyncStatus>('disabled');
  const [lastSyncedAt, setLastSyncedAt] = useState<Date | null>(null);
  const [authUser, setAuthUser] = useState<SupabaseUser | null>(null);
  const hasProcessedQueue = useRef(false);

  const isAuthenticated = authUser !== null;

  // Initialize auth state and listen for changes
  useEffect(() => {
    const supabase = createClient();
    
    // Get initial auth state
    supabase.auth.getUser().then(({ data }) => {
      setAuthUser(data.user);
      if (data.user) {
        setSyncStatus(isOnline ? 'pending' : 'offline');
      } else {
        setSyncStatus('disabled');
      }
    });

    // Listen for auth changes
    const { data: listener } = supabase.auth.onAuthStateChange((event, session) => {
      setAuthUser(session?.user ?? null);
      if (session?.user) {
        setSyncStatus(isOnline ? 'pending' : 'offline');
      } else {
        setSyncStatus('disabled');
        setLastSyncedAt(null);
      }
    });

    return () => {
      listener.subscription.unsubscribe();
    };
  }, [isOnline]);

  // Process offline queue when coming back online
  useEffect(() => {
    if (!isOnline || !authUser || hasProcessedQueue.current) return;
    
    const processQueue = async () => {
      const queue = getQueue();
      if (queue.length === 0) {
        // No queue, just pull latest
        setSyncStatus('syncing');
        try {
          await withRetry(() => pullLatest(authUser.id));
          setSyncStatus('synced');
          setLastSyncedAt(new Date());
        } catch {
          setSyncStatus('pending');
        }
        return;
      }

      setSyncStatus('syncing');
      try {
        // Push all queued workouts
        await Promise.all(
          queue.map(id =>
            withRetry(() => pushWorkout(id, authUser.id))
              .then(() => dequeueWorkout(id))
              .catch(() => {}) // Keep in queue if failed
          )
        );
        // Then pull latest
        await withRetry(() => pullLatest(authUser.id));
        setSyncStatus('synced');
        setLastSyncedAt(new Date());
      } catch {
        setSyncStatus('pending');
      }
    };

    hasProcessedQueue.current = true;
    void processQueue();
  }, [isOnline, authUser]);

  // Reset queue processing flag when going offline
  useEffect(() => {
    if (!isOnline) {
      hasProcessedQueue.current = false;
    }
  }, [isOnline]);

  // Sync after completing a workout
  const syncAfterWorkout = useCallback(async (workoutId: string) => {
    if (!authUser) return;
    
    if (!isOnline) {
      enqueueWorkout(workoutId);
      setSyncStatus('offline');
      return;
    }

    setSyncStatus('syncing');
    try {
      await withRetry(() => pushWorkout(workoutId, authUser.id));
      setSyncStatus('synced');
      setLastSyncedAt(new Date());
    } catch {
      // Failed after 3 retries, queue for later
      enqueueWorkout(workoutId);
      setSyncStatus('pending');
    }
  }, [authUser, isOnline]);

  // Manual pull from cloud
  const pullFromCloud = useCallback(async () => {
    if (!authUser || !isOnline) return;

    setSyncStatus('syncing');
    try {
      await withRetry(() => pullLatest(authUser.id));
      setSyncStatus('synced');
      setLastSyncedAt(new Date());
    } catch {
      setSyncStatus('pending');
    }
  }, [authUser, isOnline]);

  // Upload all existing local data (first sign-in flow)
  const uploadExistingData = useCallback(async (): Promise<boolean> => {
    if (!authUser) return false;

    setSyncStatus('syncing');
    try {
      await withRetry(() => uploadAllLocalData(authUser.id));
      setSyncStatus('synced');
      setLastSyncedAt(new Date());
      return true;
    } catch {
      setSyncStatus('pending');
      return false;
    }
  }, [authUser]);

  // Sign out
  const signOut = useCallback(async () => {
    const supabase = createClient();
    await supabase.auth.signOut();
    clearQueue();
    setAuthUser(null);
    setSyncStatus('disabled');
    setLastSyncedAt(null);
  }, []);

  return {
    syncStatus,
    lastSyncedAt,
    authUser,
    isAuthenticated,
    syncAfterWorkout,
    pullFromCloud,
    uploadExistingData,
    signOut,
  };
}

'use client';

import { useState, useEffect } from 'react';
import { db } from '@/lib/db/dexie';
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogHeader,
  DialogTitle,
  DialogFooter,
} from '@/components/ui/dialog';
import { Button } from '@/components/ui/button';
import { AuthDialog } from './auth-dialog';

interface SyncNudgeProps {
  /** When true, shows nudge if user is not authenticated */
  show: boolean;
  /** Callback when user completes sign-in flow */
  onComplete?: () => void;
}

type NudgeState = 'nudge' | 'auth' | 'upload-prompt' | 'uploading' | 'done';

interface SyncNudgeWithUploadProps extends SyncNudgeProps {
  uploadExistingData: () => Promise<boolean>;
  isAuthenticated: boolean;
}

export function SyncNudge({ 
  show, 
  onComplete, 
  uploadExistingData, 
  isAuthenticated 
}: SyncNudgeWithUploadProps) {
  const [state, setState] = useState<NudgeState>('nudge');
  const [hasLocalData, setHasLocalData] = useState(false);
  const [open, setOpen] = useState(false);

  // Check for existing local workouts
  useEffect(() => {
    const checkLocalData = async () => {
      const workoutCount = await db.completedWorkouts.count();
      setHasLocalData(workoutCount > 0);
    };
    void checkLocalData();
  }, []);

  // Show nudge when requested and not authenticated
  useEffect(() => {
    if (show && !isAuthenticated) {
      setOpen(true);
      setState('nudge');
    }
  }, [show, isAuthenticated]);

  // After auth success, show upload prompt if there's local data
  useEffect(() => {
    if (isAuthenticated && state === 'auth') {
      if (hasLocalData) {
        setState('upload-prompt');
      } else {
        setState('done');
        onComplete?.();
        setOpen(false);
      }
    }
  }, [isAuthenticated, state, hasLocalData, onComplete]);

  const handleEnableSync = () => {
    setState('auth');
  };

  const handleUpload = async () => {
    setState('uploading');
    const success = await uploadExistingData();
    setState('done');
    if (success) {
      onComplete?.();
    }
    setOpen(false);
  };

  const handleSkipUpload = () => {
    setState('done');
    onComplete?.();
    setOpen(false);
  };

  const handleDismiss = () => {
    setOpen(false);
    setState('nudge');
  };

  if (state === 'auth') {
    return (
      <AuthDialog
        open={open}
        onOpenChange={(isOpen) => {
          if (!isOpen) {
            handleDismiss();
          }
        }}
      />
    );
  }

  return (
    <Dialog open={open} onOpenChange={setOpen}>
      <DialogContent className="sm:max-w-md">
        {state === 'nudge' && (
          <>
            <DialogHeader>
              <DialogTitle>Keep your progress safe</DialogTitle>
              <DialogDescription>
                Enable cloud sync to back up your workout history and access it on any device.
              </DialogDescription>
            </DialogHeader>
            <DialogFooter className="flex-col gap-2 sm:flex-col">
              <Button onClick={handleEnableSync} className="w-full">
                Enable sync
              </Button>
              <Button variant="ghost" onClick={handleDismiss} className="w-full">
                Maybe later
              </Button>
            </DialogFooter>
          </>
        )}

        {state === 'upload-prompt' && (
          <>
            <DialogHeader>
              <DialogTitle>Upload existing workouts?</DialogTitle>
              <DialogDescription>
                You have workout history on this device. Would you like to upload it to the cloud?
              </DialogDescription>
            </DialogHeader>
            <DialogFooter className="flex-col gap-2 sm:flex-col">
              <Button onClick={handleUpload} className="w-full">
                Yes, upload my workouts
              </Button>
              <Button variant="outline" onClick={handleSkipUpload} className="w-full">
                No, start fresh
              </Button>
            </DialogFooter>
          </>
        )}

        {state === 'uploading' && (
          <>
            <DialogHeader>
              <DialogTitle>Uploading...</DialogTitle>
              <DialogDescription>
                Your workout history is being uploaded to the cloud.
              </DialogDescription>
            </DialogHeader>
            <div className="flex justify-center py-4">
              <div className="h-8 w-8 animate-spin rounded-full border-4 border-primary border-t-transparent" />
            </div>
          </>
        )}
      </DialogContent>
    </Dialog>
  );
}

const UserContext = createContext<UserContextValue | undefined>(undefined);

export function UserProvider({ children }: { children: React.ReactNode }) {
  // Local state
  const [user, setUser] = useState<User | null>(null);
  const [oneRepMaxes, setOneRepMaxes] = useState<OneRepMax[]>([]);
  
  // Auth & sync state
  const [authUser, setAuthUser] = useState<SupabaseUser | null>(null);
  const [syncStatus, setSyncStatus] = useState<SyncStatus>('disabled');
  const [lastSyncedAt, setLastSyncedAt] = useState<Date | null>(null);

    const userData = await getUser();
    if (userData) {
      setUser(userData);
      const orms = await getOneRepMaxesByUser(userData.id);

  // Initialize local data
  useEffect(() => {
    let isActive = true;

    void (async () => {
      const userData = await getUser();
      if (!isActive || !userData) return;

      const orms = await getOneRepMaxesByUser(userData.id);
      if (!isActive) return;

      setUser(userData);
      setOneRepMaxes(orms);
    })();

    return () => {
      isActive = false;
    };
  }, []);

  // Initialize auth state and listen for changes
  useEffect(() => {
    const supabase = createClient();
    
    // Get initial auth state
    supabase.auth.getUser().then(({ data }) => {
      setAuthUser(data.user);
      if (data.user) {
        setSyncStatus(isOnline ? 'pending' : 'offline');
      }
    });

    // Listen for auth changes
    const { data: listener } = supabase.auth.onAuthStateChange((event, session) => {
      setAuthUser(session?.user ?? null);
      if (session?.user) {
        setSyncStatus(isOnline ? 'pending' : 'offline');
      } else {
        setSyncStatus('disabled');
        setLastSyncedAt(null);
      }
    });

    return () => {
      listener.subscription.unsubscribe();
    };
  }, [isOnline]);

  // Pull on app foreground when authenticated
  useEffect(() => {
    if (!authUser || !isOnline) return;

    const handleVisibilityChange = () => {
      if (document.visibilityState === 'visible') {
        void pullFromCloud();
      }
    };

    document.addEventListener('visibilitychange', handleVisibilityChange);
    return () => {
      document.removeEventListener('visibilitychange', handleVisibilityChange);
    };
  }, [authUser, isOnline]);

  // Process offline queue when coming back online
  useEffect(() => {
    if (!isOnline || !authUser) return;

    const processQueue = async () => {
      const queue = getQueue();
      if (queue.length === 0) return;

      setSyncStatus('syncing');
      try {
        await Promise.all(
          queue.map(id =>
            withRetry(() => pushWorkout(id, authUser.id))
              .then(() => dequeueWorkout(id))
              .catch(() => {})
          )
        );
        await withRetry(() => pullLatest(authUser.id));
        setSyncStatus('synced');
        setLastSyncedAt(new Date());
        await loadUserData(); // Refresh local state
      } catch {
        setSyncStatus('pending');
      }
    };

    void processQueue();
  }, [isOnline, authUser]);

  // Local data operations
  async function updateUserProfile(updates: Partial<User>) {
    if (!user) return;
    await updateUser(user.id, updates);
    setUser(prev => prev ? { ...prev, ...updates } : null);
  }

  async function update1RM(exerciseId: string, weight: number) {
    if (!user) return;

    const new1RM: OneRepMax = {
      id: generateId(),
      userId: user.id,
      exerciseId,
      weight,
      date: new Date()
    };

    await saveOneRepMax(new1RM);
    setOneRepMaxes(prev => [...prev, new1RM]);
  }

  async function refresh() {
    await loadUserData();
  }

  // Sync operations
  const syncAfterWorkout = useCallback(async (workoutId: string) => {
    if (!authUser) return;
    
    if (!isOnline) {
      enqueueWorkout(workoutId);
      setSyncStatus('offline');
      return;
    }

    setSyncStatus('syncing');
    try {
      await withRetry(() => pushWorkout(workoutId, authUser.id));
      setSyncStatus('synced');
      setLastSyncedAt(new Date());
    } catch {
      enqueueWorkout(workoutId);
      setSyncStatus('pending');
    }
  }, [authUser, isOnline]);

  const pullFromCloud = useCallback(async () => {
    if (!authUser || !isOnline) return;

    setSyncStatus('syncing');
    try {
      await withRetry(() => pullLatest(authUser.id));
      setSyncStatus('synced');
      setLastSyncedAt(new Date());
      await loadUserData();
    } catch {
      setSyncStatus('pending');
    }
  }, [authUser, isOnline]);

  const uploadExistingData = useCallback(async (): Promise<boolean> => {
    if (!authUser) return false;

    setSyncStatus('syncing');
    try {
      await withRetry(() => uploadAll(authUser.id));
      setSyncStatus('synced');
      setLastSyncedAt(new Date());
      return true;
    } catch {
      setSyncStatus('pending');
      return false;
    }
  }, [authUser]);

  const signOut = useCallback(async () => {
    const supabase = createClient();
    await supabase.auth.signOut();
    clearQueue();
    setAuthUser(null);
    setSyncStatus('disabled');
    setLastSyncedAt(null);
  }, []);

  return (
    <UserContext.Provider value={{
      user,
      oneRepMaxes,
      updateUserProfile,
      update1RM,
      refresh,
      authUser,
      isAuthenticated,
      syncStatus,
      lastSyncedAt,
      isOnline,
      syncAfterWorkout,
      pullFromCloud,
      uploadExistingData,
      signOut,
    }}>
      {children}
    </UserContext.Provider>
  );
}

export function useUser() {
  const context = useContext(UserContext);
  if (!context) {
    throw new Error('useUser must be used within UserProvider');
  }
  return context;
}
